```python
# win7_compat.py - DPI设置
@classmethod
def setup_dpi_awareness(cls) -> bool:
    if sys.platform != 'win32':
        return False
    try:
        ctypes.windll.user32.SetProcessDPIAware()  # ✅ Windows 7兼容
        return True
    except (AttributeError, OSError):
        return False

# win7_compat.py - TLS支持
@classmethod
def create_ssl_context(cls) -> Optional['ssl.SSLContext']:
    ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_CLIENT)
    ctx.minimum_version = ssl.TLSVersion.TLSv1_2  # ✅ Windows 7 SP1支持
    ctx.check_hostname = False
    ctx.verify_mode = ssl.CERT_NONE
    return ctx
```
